package 'Distiller Project' {
    package 'Distiller Behavior' {
        private import 'Item Types'::*;
        private import Metaobjects::SemanticMetadata;
        metadata def <Continuous> Continuous :> SemanticMetadata {
            :> annotatedElement : SysML::Flow;
        }
        action def 'Heat Water' {
            doc /* Heat Water：加热进入蒸馏器的水 */
            in coldDirty : H2O;
            in recovered : Heat;
            out hotDirty : H2O;
        }
        action def 'Boil Water' {
            doc /* Boil Water：煮沸水 */
            in hotDirty : H2O;
            in external : Heat;
            out steam : H2O;
            out preDischarge : Residue;
        }
        action def 'Condense Steam' {
            doc /* 将蒸汽冷凝成液态水，并回收热量  */
            in steam : H2O;
            out pure : H2O;
            out recovered : Heat;
        }
        action def 'Drain Residue' {
            doc /* 排出蒸馏过程中产生的残渣 */
            in preDischarge : Residue;
            out discharge : Residue;
        }
        action def 'Distill Water' {
            doc /* Distill Water：整体蒸馏过程 */
            in coldDirty : H2O;
            in external : Heat {
                doc /* {stream} - 持续的热输入流 */
            }
            out pure : H2O;
            out discharge : Residue;
            action a1 : 'Heat Water';
            action a2 : 'Boil Water';
            action a3 : 'Condense Steam';
            action a4 : 'Drain Residue';
            binding bind a1.coldDirty = coldDirty;
            binding bind a2.external = external;
            binding bind pure = a3.pure;
            binding bind discharge = a4.discharge;

            #Continuous flow of2 from a1.hotDirty to a2.hotDirty;
            #Continuous flow of3 from a2.steam to a3.steam;
            #Continuous flow of5 from a2.preDischarge to a4.preDischarge;
            #Continuous flow of8 from a3.recovered to a1.recovered;
            view 'Distill Water' : DassaultSystemesViews::SymbolicViews::'Action Flow View';
        }
    }
    package 'Distiller Structure' {
        private import 'Item Types'::*;
        private import 'Distiller Behavior'::*; 

        part def 'Water Source' {
            out item waterSupply : H2O;
            doc /* 水源*/
        }
        part def 'Heat Source' {
            out item heatOut : Heat;
            doc /* 热源*/
        }

        part def HeatExchanger {
            doc /* 热交换器 */
        }

        part def Boiler {
            doc /* 煮沸器 */
        }

        part def Valve {
            doc /* 阀门 */
        }

        part def Distiller {
            in item dirtyWaterIn : H2O;
            in item heatIn : Heat;
            out item pureWaterOut : H2O;

        perform action distill : 'Distill Water';

        part condenser : HeatExchanger {
            perform distill.a1;
            perform distill.a3;
        }

        part evaporator : Boiler {
            perform distill.a2;
        }

        part drain : Valve {
            perform distill.a4;
        }
            
            doc /* 蒸馏器*/            
        }
    }
